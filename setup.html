تهيئة وتثبيت المتجر المستنسخ

قم باتباع الخطوات البسيطة التالية لربط وتثبيت قاعدة البيانات الجديدة لمتجرك.

  خطوة هامة: لكي يعمل متجرك المستنسخ بشكل صحيح، يجب أن تقوم بإنشاء الجداول اللازمة في قاعدة بيانات Supabase الخاصة بك أولاً.

قم بنسخ كود الـ SQL البرمجي أدناه بالكامل، ثم اذهب إلى حسابك في Supabase، وادخل إلى SQL Editor، وافتح ورقة عمل جديدة (New Query)، ثم الصق الكود واضغط على زر Run لتأسيس الجداول في 10 ثوانٍ فقط! 

-- Perex Store Database Setup Script
-- Paste this script inside your Supabase SQL Editor and click 'Run'!

-- 1. Create Categories Table
create table if not exists categories (
    id text primary key,
    name text not null,
    slug text not null,
    "order" integer default 0,
    is_visible boolean default true,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Create Products Table
create table if not exists products (
    id bigint generated by default as identity primary key,
    name text not null,
    price numeric not null,
    old_price numeric,
    stock integer default 0,
    badge text,
    description text,
    rating numeric default 5,
    pixel_id text,
    images text[] default '{}'::text[],
    has_colors boolean default false,
    colors text[] default '{}'::text[],
    has_sizes boolean default false,
    sizes text[] default '{}'::text[],
    is_visible boolean default true,
    category text references categories(id) on delete set null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 3. Create Coupons Table
create table if not exists coupons (
    id bigint generated by default as identity primary key,
    code text unique not null,
    type text default 'percentage',
    discount numeric not null,
    min_order numeric default 0,
    max_discount numeric default 0,
    free_shipping boolean default false,
    apply_type text default 'all',
    target_ids text[] default '{}'::text[],
    expiry_date date,
    max_uses integer default 100,
    current_uses integer default 0,
    is_active boolean default true,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 4. Create Orders Table
create table if not exists orders (
    id bigint primary key,
    status text default 'new',
    customer_name text,
    customer_phone text,
    governorate text,
    district text,
    address text,
    notes text,
    items jsonb default '[]'::jsonb,
    subtotal numeric default 0,
    shipping numeric default 0,
    discount numeric default 0,
    total numeric default 0,
    coupon text,
    admin_note text,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 5. Create Settings Table
create table if not exists settings (
    key text primary key,
    value jsonb,
    updated_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 6. Create Shipping Rates Table
create table if not exists shipping_rates (
    id bigint generated by default as identity primary key,
    name text not null,
    cost numeric not null,
    is_active boolean default true,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 7. Create Reviews Table
create table if not exists reviews (
    id bigint generated by default as identity primary key,
    product_id bigint references products(id) on delete cascade,
    customer_name text not null,
    rating integer default 5,
    comment text,
    is_visible boolean default true,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Disable Row Level Security (RLS) for simple client-side access
alter table categories disable row level security;
alter table products disable row level security;
alter table coupons disable row level security;
alter table orders disable row level security;
alter table settings disable row level security;
alter table shipping_rates disable row level security;
alter table reviews disable row level security;
